cmake_minimum_required(VERSION 3.24)
project(NexusVdeviceBridge LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build this bridge with Windows x64 Visual Studio C++ tools.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(NexusVdeviceBridge SHARED Bridge.cpp vendor/Vdevice.cpp)
target_compile_features(NexusVdeviceBridge PRIVATE cxx_std_17)
target_compile_options(NexusVdeviceBridge PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
target_compile_definitions(NexusVdeviceBridge PRIVATE UNICODE _UNICODE)
set_target_properties(NexusVdeviceBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
